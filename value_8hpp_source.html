78 template <IsAnUnderLyingValueType T> [[nodiscard]] auto as() const -> T const & {
79 if constexpr (std::is_same_v<T, bool>) {
80 return m_underlying_value.m_bool;
81 } else if constexpr (std::is_same_v<T, double>) {
82 return m_underlying_value.m_number;
83 } else if constexpr (std::is_same_v<T, Object *>) {

94 friend auto operator<<(std::ostream & os, Value const & value) -> std::ostream & {
95 switch (value.m_type) {
96 case Value::Type::BOOL:
97 return os << (value.m_underlying_value.m_bool ? "true" : "false");
98 case Value::Type::NULL_:
99 return os << "null";
100 case Value::Type::NUMBER:
101 {
102 // Remove trailing zeros and decimal point if there are no fractional digits
103 std::string str = std::to_string(value.m_underlying_value.m_number);
104 str.erase(str.find_last_not_of('0') + 1, std::string::npos);
105 str.erase(str.find_last_not_of('.') + 1, std::string::npos);
106 return os << str;
107 }
108 case Value::Type::OBJECT:
109 return os << value.m_underlying_value.m_object;
110 }
111 // should be be unreachable
112 return os << "undefined";
113 }
