40inline auto std::formatter<cppLox::Types::Object *>::format(cppLox::Types::Object * object,
41 format_context & ctx) const {
42 switch (object->type()) {
43 case cppLox::Types::Object::Type::STRING:
44 return formatter<string_view>::format(std::format("{}", object->as<cppLox::Types::ObjectString>()->string()),
45 ctx);
46 case cppLox::Types::Object::Type::FUNCTION:
47 {
48 cppLox::Types::ObjectString * name = object->as<cppLox::Types::ObjectFunction>()->name();
49 if (name != nullptr) {
50 return formatter<string_view>::format(std::format("<fn {}>", name->string()), ctx);
51 }
52 return formatter<string_view>::format(std::format("<script>"), ctx);
53 }
54 case cppLox::Types::Object::Type::NATIVE_FUNCTION:
55 {
56 return formatter<string_view>::format(std::format("<native fn>"), ctx);
57 }
58 case cppLox::Types::Object::Type::CLOSURE:
59 {
60 cppLox::Types::ObjectString * name = object->as<cppLox::Types::ObjectClosure>()->function()->name();
61 if (name != nullptr) {
62 return formatter<string_view>::format(std::format("<fn {}>", name->string()), ctx);
63 }
64 return formatter<string_view>::format(std::format("<script>"), ctx);
65 }
66 case cppLox::Types::Object::Type::UPVALUE:
67 {
68 return formatter<string_view>::format(std::format("upvalue"), ctx);
69 }
70 // should be be unreachable
71 default:
72 return formatter<string_view>::format(std::format("undefined"), ctx);
73 }
