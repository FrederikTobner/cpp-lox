44template <auto METHOD, typename EXPECTED_SIGNATURE> struct MethodSignatureCheck {
45 static constexpr auto method = METHOD;
46
47 static_assert(
48 requires { method; }, "Required method is missing");
49
50 static_assert(std::is_member_function_pointer_v<decltype(METHOD)>, "Method must be a member function pointer");
51 static_assert(std::is_member_function_pointer_v<EXPECTED_SIGNATURE>,
52 "Expected signature must be a member function pointer");
53
54 using actual_traits = MemberFunctionTraits<decltype(METHOD)>;
55 using expected_traits = MemberFunctionTraits<EXPECTED_SIGNATURE>;
56
57 static_assert(std::is_same_v<typename actual_traits::return_type, typename expected_traits::return_type>,
58 "Return type mismatch");
59 static_assert(actual_traits::arity == expected_traits::arity, "Arity mismatch");
60 static_assert(std::is_same_v<typename actual_traits::args_tuple, typename expected_traits::args_tuple>,
61 "Parameter types mismatch");
62
63 static constexpr bool allChecksSucceded = true;
64};
