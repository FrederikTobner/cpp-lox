75template <typename PARSER>
76 requires PrattParserTrait<PARSER>::hasRequiredMethods
77auto ParseRuleLookupTableFactory<PARSER>::createParseRules() -> ParseRuleLookupTable {
78 ParseRuleLookupTable rules{};
79
80 initializeGroupingTokens(rules);
81 initializeSeparators(rules);
82 initializeArithmeticOperators(rules);
83 initializeComparisonOperators(rules);
84 initializeLiteralsAndIdentifiers(rules);
85 initializeKeywordsAndLogicalOperators(rules);
86 initializeStatementKeywords(rules);
87
88 rules[static_cast<size_t>(Token::Type::END_OF_FILE)] = ParseRuleBuilder<PARSER>::nullRule();
89
90 return rules;
91}

94 requires PrattParserTrait<PARSER>::hasRequiredMethods
95auto ParseRuleLookupTableFactory<PARSER>::initializeGroupingTokens(ParseRuleLookupTable & rules) -> void {
96 rules[static_cast<size_t>(Token::Type::LEFT_PARENTHESES)] =
97 ParseRuleBuilder<PARSER>::fullRule(&PARSER::grouping, &PARSER::call, Precedence::CALL);
98 rules[static_cast<size_t>(Token::Type::RIGHT_PARENTHESES)] = ParseRuleBuilder<PARSER>::nullRule();
99 rules[static_cast<size_t>(Token::Type::LEFT_BRACE)] = ParseRuleBuilder<PARSER>::nullRule();
100 rules[static_cast<size_t>(Token::Type::RIGHT_BRACE)] = ParseRuleBuilder<PARSER>::nullRule();
101}

114void ParseRuleLookupTableFactory<PARSER>::initializeArithmeticOperators(ParseRuleLookupTable & rules) {
115 rules[static_cast<size_t>(Token::Type::MINUS)] =
116 ParseRuleBuilder<PARSER>::fullRule(&PARSER::unary, &PARSER::binary, Precedence::TERM);
117 rules[static_cast<size_t>(Token::Type::PLUS)] =
118 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::TERM);
119 rules[static_cast<size_t>(Token::Type::SLASH)] =
120 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::FACTOR);
121 rules[static_cast<size_t>(Token::Type::STAR)] =
122 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::FACTOR);

125template <typename PARSER>
126 requires PrattParserTrait<PARSER>::hasRequiredMethods
127void ParseRuleLookupTableFactory<PARSER>::initializeComparisonOperators(ParseRuleLookupTable & rules) {
128 rules[static_cast<size_t>(Token::Type::BANG)] = ParseRuleBuilder<PARSER>::prefixRule(&PARSER::unary);
129 rules[static_cast<size_t>(Token::Type::BANG_EQUAL)] =
130 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::EQUALITY);
131 rules[static_cast<size_t>(Token::Type::EQUAL)] = ParseRuleBuilder<PARSER>::nullRule();
132 rules[static_cast<size_t>(Token::Type::EQUAL_EQUAL)] =
133 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::EQUALITY);
134 rules[static_cast<size_t>(Token::Type::GREATER)] =
135 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::COMPARISON);
136 rules[static_cast<size_t>(Token::Type::GREATER_EQUAL)] =
137 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::COMPARISON);
138 rules[static_cast<size_t>(Token::Type::LESS)] =
139 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::COMPARISON);
140 rules[static_cast<size_t>(Token::Type::LESS_EQUAL)] =
141 ParseRuleBuilder<PARSER>::infixRule(&PARSER::binary, Precedence::COMPARISON);
142}

152template <typename PARSER>
153 requires PrattParserTrait<PARSER>::hasRequiredMethods
154void ParseRuleLookupTableFactory<PARSER>::initializeKeywordsAndLogicalOperators(ParseRuleLookupTable & rules) {
155 rules[static_cast<size_t>(Token::Type::AND)] = ParseRuleBuilder<PARSER>::infixRule(&PARSER::and_, Precedence::AND);
156 rules[static_cast<size_t>(Token::Type::OR)] = ParseRuleBuilder<PARSER>::infixRule(&PARSER::or_, Precedence::OR);
157 auto const literals = {Token::Type::FALSE, Token::Type::NULL_, Token::Type::TRUE};
158 for (auto type : literals) {
159 rules[static_cast<size_t>(type)] = ParseRuleBuilder<PARSER>::prefixRule(&PARSER::literal);
160 }
161}

163template <typename PARSER>
164 requires PrattParserTrait<PARSER>::hasRequiredMethods
165void ParseRuleLookupTableFactory<PARSER>::initializeStatementKeywords(ParseRuleLookupTable & rules) {
166 auto const nullKeyWords = {Token::Type::CLASS, Token::Type::ELSE, Token::Type::FUN, Token::Type::FOR,
167 Token::Type::IF, Token::Type::RETURN, Token::Type::SUPER, Token::Type::THIS,
168 Token::Type::VAR, Token::Type::WHILE};
169
170 for (auto type : nullKeyWords) {
171 rules[static_cast<size_t>(type)] = ParseRuleBuilder<PARSER>::nullRule();
172 }
173 rules[static_cast<size_t>(Token::Type::PRINT)] = ParseRuleBuilder<PARSER>::prefixRule(&PARSER::printStatement);
174}
