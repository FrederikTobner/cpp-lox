62 template <cppLox::Types::DerivedFromObject T, class... Args>
63 auto create(Args &&... args) -> cppLox::Types::Object * {
64 cppLox::Types::Object * object;
65 if constexpr (std::is_same_v<T, cppLox::Types::ObjectString>) {
66 auto string = new T(std::forward<Args>(args)...);
67 auto iterator = this->m_strings.find(string);
68 if (iterator != m_strings.end()) {
69 delete string;
70 return static_cast<cppLox::Types::Object *>(*iterator);
71 }
72 m_strings.insert(string);
73 object = static_cast<cppLox::Types::Object *>(string);
74 } else {
75 object = static_cast<cppLox::Types::Object *>(new T(std::forward<Args>(args)...));
76 }
77
78 m_objects.push_back(std::unique_ptr<cppLox::Types::Object>(object));
79 return object;
80 }
